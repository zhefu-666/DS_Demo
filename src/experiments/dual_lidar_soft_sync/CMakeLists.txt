cmake_minimum_required(VERSION 3.8)
project(dual_lidar_soft_sync)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(message_filters REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(soft_sync_demo src/soft_sync_demo.cpp)
target_compile_features(soft_sync_demo PUBLIC cxx_std_17)
ament_target_dependencies(soft_sync_demo rclcpp message_filters livox_ros_driver2)
install(TARGETS soft_sync_demo DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/summarize.py scripts/run_mock_tests.py DESTINATION lib/${PROJECT_NAME})
ament_package()
